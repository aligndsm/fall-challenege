tasks: {}

// Load initial local state
function loadLocalState() {
const taskElements = document.querySelectorAll('.task-box');
taskElements.forEach(box => {
const taskId = box.dataset.id;
const savedVal = localStorage.getItem(`align-task-${taskId}`);
state.tasks[taskId] = savedVal === 'true';
});
updateUI();
}

// Save local state helper
function saveLocalState() {
Object.keys(state.tasks).forEach(taskId => {
localStorage.setItem(`align-task-${taskId}`, state.tasks[taskId]);
});
}

// UI Refresh Logic
function updateUI() {
let totalChecked = 0;
const totalTasks = 12;

// Update Task Boxes
const taskElements = document.querySelectorAll('.task-box');
taskElements.forEach(box => {
const taskId = box.dataset.id;
const isChecked = !!state.tasks[taskId];
const badge = box.querySelector('.check-badge');

if (isChecked) {
box.classList.add('is-checked');
badge.classList.remove('opacity-0', 'scale-50');
badge.classList.add('opacity-100', 'scale-100');
totalChecked++;
} else {
box.classList.remove('is-checked');
badge.classList.add('opacity-0', 'scale-50');
badge.classList.remove('opacity-100', 'scale-100');
}
});

// Update Week Complete Badges
[1, 2, 3].forEach(weekNum => {
const weekTasks = [`w${weekNum}-1`, `w${weekNum}-2`, `w${weekNum}-3`, `w${weekNum}-4`].map(id => !!state.tasks[id]);
const isWeekComplete = weekTasks.every(Boolean);
const completeBox = document.querySelector(`[data-week-complete="${weekNum}"]`);
const completeIcon = completeBox.querySelector('.complete-icon');

if (isWeekComplete) {
completeBox.classList.add('all-complete');
completeIcon.className = 'complete-icon fa-solid fa-circle-check text-align-gold text-2xl mb-1';
} else {
completeBox.classList.remove('all-complete');
completeIcon.className = 'complete-icon fa-solid fa-trophy text-gray-500 text-lg mb-1';
}
});

// Update Progress Bar
const percent = Math.round((totalChecked / totalTasks) * 100);
document.getElementById('progressPercent').innerText = `${percent}%`;
document.getElementById('progressBar').style.width = `${percent}%`;
}

// Trigger Confetti Celebration
function triggerConfetti() {
if (typeof confetti === 'function') {
confetti({
particleCount: 80,
spread: 70,
origin: { y: 0.6 },
colors: ['#dfcaad', '#ffffff', '#e6ded3', '#d4af37']
});
}
}

// Task Click Handler
function toggleTask(taskId) {
const wasChecked = !!state.tasks[taskId];
state.tasks[taskId] = !wasChecked;
saveLocalState();
updateUI();
saveToFirestore();

if (!wasChecked) {
const weekNum = taskId.charAt(1);
const weekTasks = [`w${weekNum}-1`, `w${weekNum}-2`, `w${weekNum}-3`, `w${weekNum}-4`].map(id => !!state.tasks[id]);
if (weekTasks.every(Boolean)) {
triggerConfetti();
}
}
}

// Complete Week Box Handler (Toggle whole week)
function toggleWeek(weekNum) {
const weekTaskIds = [`w${weekNum}-1`, `w${weekNum}-2`, `w${weekNum}-3`, `w${weekNum}-4`];
const allCurrentlyChecked = weekTaskIds.every(id => !!state.tasks[id]);
weekTaskIds.forEach(id => {
state.tasks[id] = !allCurrentlyChecked;
});

saveLocalState();
updateUI();
saveToFirestore();

if (!allCurrentlyChecked) {
triggerConfetti();
}
}

// Firestore Save Logic
async function saveToFirestore() {
if (!isFirebaseAvailable || !currentUser) return;
try {
const docRef = doc(db, 'artifacts', appId, 'users', currentUser.uid, 'challengeProgress', 'weekData');
await setDoc(docRef, {
tasks: state.tasks,
updatedAt: new Date().toISOString()
}, { merge: true });
} catch (err) {
console.warn('Firestore sync failed, local state preserved:', err);
}
}

// Switch Auth Modal Mode (Sign In vs Sign Up)
function setAuthMode(mode) {
authMode = mode;
const title = document.getElementById('authModalTitle');
const submitBtn = document.getElementById('authSubmitBtn');
const tabSignIn = document.getElementById('tabSignIn');
const tabSignUp = document.getElementById('tabSignUp');
const nameGroup = document.getElementById('nameFieldGroup');
const authError = document.getElementById('authError');

authError.classList.add('hidden');

if (mode === 'signup') {
title.innerText = 'Create Account';
submitBtn.innerText = 'Create Account';
tabSignUp.className = 'flex-1 py-2 text-xs font-semibold uppercase tracking-wider rounded-md bg-align-boxBg text-align-gold transition-all';
tabSignIn.className = 'flex-1 py-2 text-xs font-semibold uppercase tracking-wider rounded-md text-gray-400 hover:text-white transition-all';
nameGroup.classList.remove('hidden');
} else {
title.innerText = 'Sign In';
submitBtn.innerText = 'Sign In';
tabSignIn.className = 'flex-1 py-2 text-xs font-semibold uppercase tracking-wider rounded-md bg-align-boxBg text-align-gold transition-all';
tabSignUp.className = 'flex-1 py-2 text-xs font-semibold uppercase tracking-wider rounded-md text-gray-400 hover:text-white transition-all';
nameGroup.classList.add('hidden');
}
}

function showAuthError(msg) {
const errDiv = document.getElementById('authError');
errDiv.innerText = msg;
errDiv.classList.remove('hidden');
}

// Firebase Auth & Realtime Sync Setup
async function initFirebase() {
try {
app = initializeApp(firebaseConfig);
auth = getAuth(app);
db = getFirestore(app);

// Initial silent sign-in fallback if no active user session
const initialAuthToken = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;
if (initialAuthToken) {
await signInWithCustomToken(auth, initialAuthToken);
}

onAuthStateChanged(auth, (user) => {
if (unsubscribeFirestore) {
unsubscribeFirestore();
unsubscribeFirestore = null;
}

if (user) {
currentUser = user;
isFirebaseAvailable = true;

const emailDisplay = user.displayName || user.email || (user.isAnonymous ? 'Guest' : 'Member');
document.getElementById('userEmailText').innerText = emailDisplay;
if (user.isAnonymous) {
document.getElementById('authBtnText').innerText = 'Sign In / Register';
} else {
document.getElementById('authBtnText').innerText = 'Sign Out';
}

document.getElementById('syncStatus').innerHTML = `
<span class="w-2.5 h-2.5 rounded-full bg-emerald-500"></span>
Cloud Synced
`;

// Subscribe to live Firestore updates for this user
const docRef = doc(db, 'artifacts', appId, 'users', user.uid, 'challengeProgress', 'weekData');
unsubscribeFirestore = onSnapshot(docRef, (snapshot) => {
if (snapshot.exists()) {
const data = snapshot.data();
if (data && data.tasks) {
state.tasks = { ...state.tasks, ...data.tasks };
saveLocalState();
updateUI();
}
}
}, (err) => {
console.warn("Firestore snapshot error:", err);
});
} else {
// Automatically sign in anonymously as fallback
signInAnonymously(auth).catch(err => console.warn(err));
}
});
} catch (err) {
console.warn("Operating in offline/local storage mode:", err);
document.getElementById('syncStatus').innerHTML = `
<span class="w-2.5 h-2.5 rounded-full bg-amber-500"></span>
Local Storage
`;
}
}

// Event Listeners Registration
document.addEventListener('DOMContentLoaded', () => {

initFirebase();

// Task Box Clicks
document.querySelectorAll('.task-box').forEach(box => {
box.addEventListener('click', () => {
toggleTask(box.dataset.id);

});

// Week Complete Box Clicks
document.querySelectorAll('.complete-box').forEach(box => {
box.addEventListener('click', () => {
const weekNum = box.dataset.weekComplete;
toggleWeek(weekNum);
});
});

// Auth Modal Controls
const authModal = document.getElementById('authModal');
const authModalBtn = document.getElementById('authModalBtn');
const closeAuthModal = document.getElementById('closeAuthModal');
const guestContinueBtn = document.getElementById('guestContinueBtn');

authModalBtn.addEventListener('click', () => {
if (currentUser && !currentUser.isAnonymous) {
// Sign out action
signOut(auth).then(() => {
document.getElementById('userEmailText').innerText = 'Guest';
document.getElementById('authBtnText').innerText = 'Sign In / Register';
});
} else {
// Open Modal
authModal.classList.remove('hidden');
}
});

closeAuthModal.addEventListener('click', () => authModal.classList.add('hidden'));
guestContinueBtn.addEventListener('click', () => authModal.classList.add('hidden'));

document.getElementById('tabSignIn').addEventListener('click', () => setAuthMode('signin'));
document.getElementById('tabSignUp').addEventListener('click', () => setAuthMode('signup'));

// Handle Auth Form Submission
document.getElementById('authForm').addEventListener('submit', async (e) => {
e.preventDefault();
const email = document.getElementById('authEmail').value.trim();
const password = document.getElementById('authPassword').value;
const name = document.getElementById('authName').value.trim();

try {
if (authMode === 'signup') {
const userCred = await createUserWithEmailAndPassword(auth, email, password);
if (name && userCred.user) {
await updateProfile(userCred.user, { displayName: name });
}
} else {
await signInWithEmailAndPassword(auth, email, password);
}
authModal.classList.add('hidden');
document.getElementById('authForm').reset();
} catch (err) {
console.error("Auth error:", err);
let readableError = "Authentication failed. Please check your details.";
if (err.code === 'auth/email-already-in-use') readableError = "Email is already registered. Try signing in.";
if (err.code === 'auth/wrong-password' || err.code === 'auth/invalid-credential') readableError = "Invalid email or password.";
if (err.code === 'auth/weak-password') readableError = "Password should be at least 6 characters.";
showAuthError(readableError);
}
});

// Reset Progress Button
document.getElementById('resetBtn').addEventListener('click', () => {
if (confirm("Are you sure you want to reset all progress for your challenge board?")) {
Object.keys(state.tasks).forEach(id => {
state.tasks[id] = false;
});
saveLocalState();
updateUI();
saveToFirestore();
}
});
});
